# docker-compose.yml ဖိုင်တစ်ခု၏ ဖွဲ့စည်းပုံ

Compose ဖိုင်တစ်ခုတွင် အဓိက အပိုင်း သုံးပိုင်း ပါဝင်သည်: `services`၊ `networks` နှင့် `volumes` တို့ ဖြစ်ကြသည်။ အပိုင်းတစ်ခုချင်းစီရှိ ရွေးချယ်စရာ (option) အားလုံးကို နားလည်ထားခြင်းဖြင့် ခိုင်မာပြီး Production အတွက် အသင့်ဖြစ်နေသော စတက်ခ် (stacks) များကို တည်ဆောက်နိုင်မည်ဖြစ်သည်။

```yaml
# docker-compose.yml — annotated skeleton
services:       # ကွန်တိန်နာ တစ်ခုချင်းစီကို ဤနေရာတွင် သတ်မှတ်ပါ (မဖြစ်မနေ လိုအပ်သည်)
  web: ...
  db: ...

networks:       # စိတ်ကြိုက် ဂျွန်ယက် သတ်မှတ်ချက်များ (ဆန္ဒအလျောက် — မပါလျှင် အလိုအလျောက် ဖန်တီးပေးသည်)
  myapp-net:
    driver: bridge

volumes:        # အမည်ပေးထားသော အချက်အလက် ထိန်းသိမ်းရာ တွဲဖက်များ (optional)
  postgres-data:
  redis-data:
```

---

## `services` အပိုင်း: သော့ချက် (Key) တစ်ခုချင်းစီ၏ ရှင်းလင်းချက်

### `image` နှင့် `build`

```yaml
services:
  # ရွေးချယ်စရာ A: Registry တစ်ခုမှ အသင့်တည်ဆောက်ပြီးသား Image ကို ဆွဲယူရန် (Pull)
  db:
    image: postgres:16-alpine          # Docker Hub image
  
  cache:
    image: redis:7-alpine
  
  private-service:
    image: ghcr.io/yourorg/app:v1.2.0  # GitHub Container Registry
    # Private registries အတွက် ဦးစွာ run ပါ: docker login ghcr.io

  # ရွေးချယ်စရာ B: ဒေသတွင်းရှိ Dockerfile မှ တည်ဆောက်ရန် (Build)
  api:
    build:
      context: .                       # Dockerfile ပါဝင်သော ဖိုင်တွဲ (Directory)
      dockerfile: Dockerfile           # ပုံမှန်အမည် — ချန်လှပ်ထားနိုင်သည်
      args:                            # တည်ဆောက်ချိန် (Build-time) ARG တန်ဖိုးများ
        NODE_VERSION: "20"
        BUILD_ENV: production
      target: runner                   # Multi-stage build: ဤအဆင့်တွင် ရပ်ရန်
      cache_from:                      # Layer cache ဖြင့် CI တည်ဆောက်မှုကို အမြန်နှုန်းမြှင့်တင်ရန်
        - type=gha                     # GitHub Actions cache
      labels:
        org.opencontainers.image.source: "https://github.com/yourorg/app"
```

### `ports` — Host နှင့် Container ပေါ့တ်ချိတ်ဆက်မှု (Port Mapping)

```yaml
services:
  api:
    ports:
      # "HOST_PORT:CONTAINER_PORT"
      - "3000:3000"             # Host ပေါ့တ် 3000 တွင် API ကို ဖွင့်လှစ်ရန်

      # သတ်မှတ်ထားသော Host IP သို့ ချိတ်ဆက်ရန် (Interfaces အားလုံးသို့ မဖွင့်ဘဲ)
      - "127.0.0.1:3000:3000"  # Localhost မှသာ ဝင်ရောက်နိုင်မည် — အင်တာနက်မှ မဟုတ်ပါ

      # Docker အား ကျပန်း (Random) Host ပေါ့တ် ရွေးချယ်ခွင့်ပြုရန် (ပဋိပକ୍ෂမှုများကို ရှောင်ရှားရန် CI တွင် အသုံးဝင်သည်)
      - "3000"                  # ကျပန်း Host ပေါ့တ် → ကွန်တိန်နာ ပေါ့တ် 3000

      # ပရိုတိုကော ပါသော အမည်ခံ ပေါ့တ်
      - "5353:53/udp"           # DNS ဝန်ဆောင်မှုများအတွက် UDP ပေါ့တ်
```

<Callout type="tip" title="Production တွင် 127.0.0.1 သို့ ချိတ်ဆက်ပါ">
ဒေတာဘေ့စ် ပေါ့တ်များ (5432, 6379, 27017) ကို အများပြည်သူသုံး ဆာဗာတစ်ခုပေါ်ရှိ `0.0.0.0` သို့ ဘယ်တော့မှ မဖွင့်ပါနှင့်။ ဒေသတွင်း (Local-only) သုံးစွဲရန်အတွက်သာ `127.0.0.1` သို့ ချိတ်ဆက်ပါ (သို့မဟုတ်) `ports` ကို လုံးဝ ချန်လှပ်ထားပြီး အတွင်းပိုင်း Docker ဂျွန်ယက်ကို အသုံးပြုပါ - တူညီသော Compose ဂျွန်ယက်ပေါ်ရှိ ကွန်တိန်နာများသည် တစ်ခုနှင့်တစ်ခု အမြဲ ချိတ်ဆက်နိုင်ကြသည်။
</Callout>

### `environment` — ပတ်ဝန်းကျင်ဆိုင်ရာ ကိန်းရှင်များ သတ်မှတ်ခြင်း (Env Vars)

```yaml
services:
  api:
    # နည်းလမ်း ၁: စာရင်း ပုံစံ (List form)
    environment:
      - NODE_ENV=production
      - PORT=3000
      - LOG_LEVEL=info

    # နည်းလမ်း ၂: Map ပုံစံ (ပို၍ ရှင်းလင်းသည်၊ ပိုမိုနှစ်သက်သည်)
    environment:
      NODE_ENV: production
      PORT: "3000"
      LOG_LEVEL: info

    # နည်းလမ်း ၃: Host ပတ်ဝန်းကျင်မှ ရယူသုံးစွဲရန် (တန်ဖိုးသည် သင့် Shell မှ လာမည်)
    environment:
      - DATABASE_URL       # အကယ်၍ DATABASE_URL ကို သင့် Shell တွင် သတ်မှတ်ထားလျှင် ၎င်းကို ဖြတ်သန်းယူမည်
      - API_KEY            # = မပါပါက: Host ၏ တန်ဖိုးကို အသုံးပြုမည်

    # နည်းလမ်း ၄: ကိန်းရှင် ထည့်သွင်းခြင်း (Variable interpolation)မှတဆင့် .env ဖိုင် တန်ဖိုးများကို အသုံးပြုရန်
    environment:
      DATABASE_URL: "postgres://${DB_USER}:${DB_PASS}@db:5432/${DB_NAME}"
```

### `env_file` — ဖိုင်များမှ ဖတ်ရှုတင်သွင်းရန်

```yaml
services:
  api:
    env_file:
      - .env               # .env ဖိုင်မှ KEY=VALUE တွဲများ အားလုံးကို တင်သွင်းပါ
      - .env.local         # ဒေသတွင်း သတ်မှတ်ချက်များဖြင့် အစားထိုးရန် (.gitignore တွင် ထည့်ထားသော)
```

### `volumes` — Bind Mounts နှင့် အမည်ပေးထားသော တွဲဖက်များ (Named Volumes)

```yaml
services:
  api:
    volumes:
      # Bind mount: host_path:container_path
      - ./src:/app/src                   # Development တွင် Hot reload လုပ်ရန်
      - ./config.json:/app/config.json   # ဖိုင်တစ်ခုတည်းကို Mount လုပ်ရန်

      # Named volume: volume_name:container_path
      - node_modules:/app/node_modules   # ကွန်တိန်နာအတွင်း node_modules ကို သီးသန့်ခွဲထားရန်

      # ဖတ်ရန်သက်သက် (Read-only) mount
      - ./certs:/etc/ssl/certs:ro        # ro = read-only

  db:
    volumes:
      # အမြဲတမ်း သိမ်းဆည်းမည့် အချက်အလက်များ — ဒေတာဘေ့စ်များအတွက် Named volumes ကို အမြဲသုံးပါ
      - postgres-data:/var/lib/postgresql/data
```

### `depends_on` — မှီခိုမှု အစဉ်လိုက် သတ်မှတ်ခြင်း (Dependency Ordering)

```yaml
services:
  api:
    depends_on:
      # ရိုးရှင်းသော ပုံစံ: ကွန်တိန်နာ စတင်ရန် (START) ကိုသာ စောင့်ရန် (ကျန်းမာရေးစစ်ဆေးမှု ပြီးသည်အထိ မစောင့်ပါ)
      - db

      # တိုးချဲ့ထားသော ပုံစံ: သတ်မှတ်ထားသော အခြေအနေများကို စောင့်ရန်
      db:
        condition: service_healthy      # ကျန်းမာရေးစစ်ဆေးမှု (Healthcheck) အောင်မြင်သည်အထိ စောင့်ပါ
        restart: true                   # db ပြန်စပါက api ကိုပါ ပြန်စပါ
      cache:
        condition: service_started      # ကွန်တိန်နာ စတင်သည်အထိသာ စောင့်ပါ
      migrate:
        condition: service_completed_successfully  # migrate ပြီးဆုံးကာ 0 ထွက်သည်အထိ စောင့်ပါ
```

### `healthcheck` — ဝန်ဆောင်မှုတစ်ခု အမှန်တကယ် အသင့်ဖြစ်ပြီလားဆိုதை စစ်ဆေးရန်

```yaml
services:
  db:
    image: postgres:16-alpine
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U ${POSTGRES_USER} -d ${POSTGRES_DB}"]
      interval: 5s      # 5 စက္ကန့်တိုင်း စစ်ဆေးပါ
      timeout: 5s       # စစ်ဆေးရန် 5 စက္ကန့်ထက် ပိုကြာပါက မအောင်မြင်ဟု သတ်မှတ်ပါ
      retries: 10       # ဆက်တိုက် 10 ကြိမ် မအောင်မြင်မှသာ ကျန်းမာရေးမကောင်းကြောင်း (unhealthy) သတ်မှတ်ပါ
      start_period: 15s # အမှားများအဖြစ် မရေတွက်မီ ကြားကာလ (ပထမအကြိမ် စတင်မှု နှေးကွေးတတ်သည်)

  cache:
    image: redis:7-alpine
    healthcheck:
      test: ["CMD", "redis-cli", "ping"]
      interval: 5s
      timeout: 3s
      retries: 5

  api:
    build: .
    healthcheck:
      test: ["CMD-SHELL", "wget -qO- http://localhost:3000/health || exit 1"]
      interval: 30s
      timeout: 10s
      retries: 3
      start_period: 20s
```

### `restart` — ပြန်လည်စတင်ရေး မူဝါဒ (Recovery Policy)

```yaml
services:
  api:
    restart: "no"              # ဘယ်တော့မှ မစတင်ပါ (ပုံသေ - default) — တစ်ကြိမ်တည်း လုပ်ဆောင်မည့် အလုပ်များအတွက် ကောင်းသည်
    # restart: "on-failure"    # အမှားကုဒ် (Non-zero exit code) ထွက်မှသာ ပြန်စပါ
    # restart: "always"        # လူကိုယ်တိုင် ရပ်ထားလျှင်တದಾင် အမြဲ ပြန်စပါ
    # restart: "unless-stopped" # လူကိုယ်တိုင် ရပ်လိုက်သည်မှလွဲ၍ အမြဲ ပြန်စပါ
```

<Callout type="tip" title="Production တွင် unless-stopped ကို အသုံးပြုပါ">
`unless-stopped` သည် ကြာရှည်စွာ လည်ပတ်နေရသည့် Production ဝန်ဆောင်မှုများအတွက် အကောင်းဆုံး မူဝါဒဖြစ်သည်။ ၎င်းသည် ချို့ယွင်းမှုများ (Crashes) ဖြစ်ပွားပြီးနောက် နှင့် ဆာဗာကို ပြန်လည်စတင်ပြီးနောက် (Reboots) တွင် အလိုအလျောက် ပြန်စတင်ပေးသည် — သို့သော် ဝန်ဆောင်မှုကို သင်ကိုယ်တိုင် ရည်ရွယ်ချက်ရှိရှိ ရပ်တန့်လိုက်သည့်အခါ `docker compose stop` ကို လိုက်နာပေးသည်။
</Callout>

### `command` နှင့် `entrypoint` — ကွန်တိန်နာ၏ မူလသတ်မှတ်ချက်များကို ကျော်လွန်လုပ်ဆောင်ရန် (Override)

```yaml
services:
  api:
    image: node:20-alpine
    working_dir: /app
    command: ["node", "server.js"]       # Dockerfile ရှိ CMD ကို ကျော်လွန်လုပ်ဆောင်သည်

  worker:
    build: .
    entrypoint: ["/bin/sh", "-c"]        # ENTRYPOINT ကို ကျော်လွန်လုပ်ဆောင်သည်
    command: ["while true; do node worker.js; sleep 5; done"]
```

### `deploy` — အရင်းအမြစ် ကန့်သတ်ချက်များ (Resource Limits - Compose v2)

```yaml
services:
  api:
    deploy:
      resources:
        limits:
          cpus: "0.5"          # CPU တုံး (core) တစ်ခု၏ အများဆုံး 50%
          memory: "512M"       # အများဆုံး RAM 512 MB — ကျော်လွန်ပါက OOMKilled ဖြစ်မည်
        reservations:
          cpus: "0.1"          # အာမခံချက်ရှိသော အနည်းဆုံး CPU
          memory: "128M"       # အာမခံချက်ရှိသော အနည်းဆုံး RAM

      # မိတ္တူပွား အရေအတွက် (docker compose --scale က ဤအချက်ကို ကျော်လွန်လုပ်ဆောင်နိုင်သည်)
      replicas: 1

      restart_policy:
        condition: on-failure
        delay: 5s
        max_attempts: 3
        window: 120s
```

### `profiles` — ရွေးချယ်နိုင်သော ဝန်ဆောင်မှုများ

```yaml
# သတ်မှတ်ထားသော profile ကို ဖွင့်ထားမှသာ ဝန်ဆောင်မှုများ စတင်မည်
services:
  adminer:                        # ဒေတာဘေ့စ် စီမံခန့်ခွဲမှု UI
    image: adminer
    ports: ["8080:8080"]
    profiles: ["tools"]           # docker compose --profile tools up ဖြင့်သာ စတင်မည်

  mailhog:                        # အီးမေးလ် စမ်းသပ်ခြင်း
    image: mailhog/mailhog
    ports: ["1025:1025", "8025:8025"]
    profiles: ["tools"]
```

## `networks` အပိုင်း

ပုံသေအားဖြင့် Compose သည် `<project>_default` အမည်ရှိ bridge ဂျွန်ယက်တစ်ခုကို ဖန်တီးပေးပြီး ဝန်ဆောင်မှုအားလုံးကို ၎င်းနှင့် ချိတ်ဆက်ပေးသည်။ ဝန်ဆောင်မှုများသည် တစ်ခုနှင့်တစ်ခု **service name** (DNS hostname အဖြစ် အသုံးပြုသော) ဖြင့် ချိတ်ဆက်ကြသည်:

```bash
# 'api' ကွန်တိန်နာသည် 'db' ဟူသော hostname ဖြင့် postgres ကို ချိတ်ဆက်သည်
DATABASE_URL=postgres://user:pass@db:5432/myapp

# ကွန်တိန်နာအမည် သို့မဟုတ် IP လိပ်စာကို သုံးရန် မလိုအပ်ပါ
```

### သီးသန့်ခွဲထုတ်ရန်အတွက် စိတ်ကြိုက် ဂျွန်ယက်များ (Custom Networks)

```yaml
networks:
  frontend:
    driver: bridge
  backend:
    driver: bridge
  # ပြင်ပ ဂျွန်ယက် (Compose ပြင်ပတွင် ဖန်တီးထားသည် — ပရောဂျက်များကြား ဆက်သွယ်ရန် သုံးသည်)
  shared-proxy:
    external: true
    name: nginx-proxy-net

services:
  nginx:
    image: nginx:alpine
    networks:
      - frontend   # အင်တာနက်သို့ ချိတ်ဆက်နိုင်သည်
      - backend    # API သို့ ချိတ်ဆက်နိုင်သည်

  api:
    build: .
    networks:
      - backend    # db နှင့် cache သို့ ချိတ်ဆက်နိုင်သည်
                   # အင်တာနက်မှ တိုက်ရိုက် ဝင်ရောက်၍ မရပါ

  db:
    image: postgres:16-alpine
    networks:
      - backend    # backend ဂျွန်ယက်မှသာ ဝင်ရောက်နိုင်သည်
                   # nginx သို့မဟုတ် အင်တာနက်သို့ မဖွင့်ထားပါ
```

ဤသို့ သီးသန့်ခွဲခြားထားခြင်းသည် Production အတွက် **လုံခြုံရေးဆိုင်ရာ အကောင်းဆုံး ကျင့်စဉ် (security best practice)** တစ်ခု ဖြစ်သည် — တစ်စုံတစ်ဦးက `nginx` ကွန်တိန်နာကို ထိုးဖောက်ဝင်ရောက်နိုင်ခဲ့လျှင်တောင် သင်၏ ဒေတာဘေ့စ်ကို အင်တာနက်မှ လှမ်း၍ မရနိုင်ပါ။

```yaml
  shared-uploads:

      device: ":/exports/uploads"

  production-db:
    external: true             # Docker သည် ဤတွဲဖက်ကို ဖန်တီးခြင်း သို့မဟုတ် ဖျက်ဆီးခြင်း မပြုလုပ်ပါ
```

## မရှိမဖြစ် လိုအပ်သော `docker compose` အမိန့်ခိုးများ (Commands)

```bash
# === စတင်ခြင်းနှင့် ရပ်တန့်ခြင်း ===
docker compose up                    # ရှေ့ဆုံးမျက်နှာပြင်တွင် စတင်ပါ (လော့ဂ်များကို ပြသသည်)
docker compose up -d                 # နောက်ခံတွင် စတင်ပါ
docker compose up -d --build         # မစတင်မီ Image အားလုံးကို အတင်းအကျပ် ပြန်လည်တည်ဆောက်ပါ
docker compose down                  # ရပ်တန့်ပြီး ကွန်တိန်နာများကို ဖယ်ရှားပါ (တွဲဖက်များကို သိမ်းဆည်းထားသည်)
docker compose down -v               # ရပ်တန့်ပြီး ကွန်တိန်နာများနှင့် တွဲဖက်များကိုပါ ဖယ်ရှားပါ (ဒေတာများ ဆုံးရှုံးမည်!)
docker compose restart api           # api ဝန်ဆောင်မှုကိုသာ ပြန်စပါ
docker compose stop db               # မဖယ်ရှားဘဲ db ကို ရပ်တန့်ပါ
docker compose start db              # ရပ်တန့်ထားသော ဝန်ဆောင်မှုကို ပြန်စပါ

# === အတိုင်းအတာပြောင်းလဲခြင်း (Scaling) ===
docker compose up -d --scale api=3  # api ၏ မိတ္တူ 3 ခုကို ဖွင့်ပါ (Load balancer တစ်ခု လိုအပ်သည်)

# === လော့ဂ်များ (Logs) ===
docker compose logs                  # ဝန်ဆောင်မှု လော့ဂ်များ အားလုံး
docker compose logs api              # api လော့ဂ်များသာ
docker compose logs -f               # ဆက်လက်ကြည့်ရှုရန် (tail -f နှင့် တူသည်)
docker compose logs -f api --tail=50 # နောက်ဆုံး စာကြောင်း 50 နှင့် ဆက်လက်ကြည့်ရှုရန်

# === စစ်ဆေးခြင်း ===
docker compose ps                    # ကွန်တိန်နာများနှင့် ၎င်းတို့၏ အခြေအနေများကို ကြည့်ရှုရန်
docker compose ps -a                 # ရပ်တန့်သွားသော ကွန်တိန်နာများကိုပါ ထည့်သွင်းပြသရန်
docker compose top                   # ကွန်တိန်နာ တစ်ခုချင်းစီတွင် လုပ်ဆောင်နေသော ဖြစ်စဉ်များကို ပြသရန်
docker compose port api 3000         # api:3000 သို့ ချိတ်ဆက်ထားသော Host ပေါ့တ်ကို ရှာရန်

# === Exec ===
docker compose exec api /bin/sh      # လည်ပတ်နေသော api ကွန်တိန်နာအတွင်းသို့ Shell ဖြင့် ဝင်ရောက်ရန်
docker compose exec db psql -U appuser -d myapp  # db ကွန်တိန်နာအတွင်း psql ကို လုပ်ဆောင်ရန်
docker compose run --rm api env      # ကွန်တိန်နာအသစ်တစ်ခုတွင် တစ်ကြိမ်သုံး အမိန့်ခိုးကို လုပ်ဆောင်ရန်

# === တည်ဆောက်ခြင်း (Build) ===
docker compose build                 # တည်ဆောက်မှု သတ်မှတ်ချက်ပါသော ဝန်ဆောင်မှု အားလုံးကို တည်ဆောက်ပါ
docker compose build api --no-cache  # api ကို အစအဆုံး အတင်းအကျပ် ပြန်တည်ဆောက်ပါ

# === Config ===
docker compose config                # ပေါင်းစပ်ထားသော compose config ကို စစ်ဆေးပြီး ထုတ်ပြပါ
docker compose config --services     # ဝန်ဆောင်မှုအမည် အားလုံးကို စာရင်းပြုစုပါ
```

## Production အတွက် အသင့်ဖြစ်နေသော ပြီးပြည့်စုံသည့် ဥပမာ

```yaml
# docker-compose.yml — production-grade three-tier app
services:
  api:
    build:
      context: .
      dockerfile: Dockerfile
      target: runner              # Multi-stage: production အဆင့်သာ
    ports:
      - "127.0.0.1:3000:3000"    # အတွင်းပိုင်းအတွက်သာ — nginx ဖြင့် ရှေ့ဆုံးမှ ခံထားသည်
    environment:
      NODE_ENV: production
      DATABASE_URL: "postgres://${POSTGRES_USER}:${POSTGRES_PASSWORD}@db:5432/${POSTGRES_DB}"
      REDIS_URL: "redis://cache:6379"
    depends_on:
      db:
        condition: service_healthy
      cache:
        condition: service_healthy
    restart: unless-stopped
    networks:
      - backend
    deploy:
      resources:
        limits: { cpus: "0.5", memory: "512M" }
        reservations: { cpus: "0.1", memory: "128M" }
    logging:
      driver: json-file
      options:
        max-size: "10m"
        max-file: "3"
    healthcheck:
      test: ["CMD-SHELL", "wget -qO- http://localhost:3000/health || exit 1"]
      interval: 30s
      timeout: 10s
      retries: 3
      start_period: 20s

  db:
    image: postgres:16-alpine
    environment:
      POSTGRES_USER: ${POSTGRES_USER}
      POSTGRES_PASSWORD: ${POSTGRES_PASSWORD}
      POSTGRES_DB: ${POSTGRES_DB}
    volumes:
      - postgres-data:/var/lib/postgresql/data
      - ./db/init.sql:/docker-entrypoint-initdb.d/init.sql  # ပထမအကြိမ် စတင်ချိန်တွင် လုပ်ဆောင်သည်
    restart: unless-stopped
    networks:
      - backend
    deploy:
      resources:
        limits: { cpus: "1.0", memory: "1G" }
    logging:
      driver: json-file
      options:
        max-size: "10m"
        max-file: "3"
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U ${POSTGRES_USER} -d ${POSTGRES_DB}"]
      interval: 5s
      timeout: 5s
      retries: 10
      start_period: 15s

  cache:
    image: redis:7-alpine
    command: redis-server --save 60 1 --loglevel warning --maxmemory 256mb --maxmemory-policy allkeys-lru
    volumes:
      - redis-data:/data
    restart: unless-stopped
    networks:
      - backend
    deploy:
      resources:
        limits: { cpus: "0.25", memory: "300M" }
    healthcheck:
      test: ["CMD", "redis-cli", "ping"]
      interval: 5s
      timeout: 3s
      retries: 5

networks:
  backend:
    driver: bridge

volumes:
  postgres-data:
  redis-data:
```

## အနှစ်ချုပ်

Compose YAML ဖိုင်ရှိ သော့ချက် (Key) တိုင်းတွင် သီးသန့် ရည်ရွယ်ချက် ရှိကြသည်:
- **`services`**: အဓိက အပိုင်း — ကွန်တိန်နာ တစ်ခုချင်းစီကို image/build၊ ports၊ environment၊ volumes၊ dependencies၊ health checks၊ restart policies နှင့် resource limits တို့ဖြင့် သတ်မှတ်ပေးသည်
- **`image`** သည် registry မှ ဆွဲယူသည်၊ **`build`** သည် ဒေသတွင်း Dockerfile မှ တည်ဆောက်သည် (multi-stage `target` ထောက်ပံ့မှုဖြင့်)
- **`ports`** သည် host နှင့် container ကို ချိတ်ဆက်ပေးသည်၊ အများပြည်သူသုံးရန် မရည်ရွယ်သော ဝန်ဆောင်မှုများအတွက် `127.0.0.1` သို့ အမြဲ ချိတ်ဆက်ပါ
- `condition: service_healthy` ပါရှိသော **`depends_on`** သည် ဝန်ဆောင်မှုများကို မှန်ကန်သော အစဉ်လိုက် စတင်စေရန် သေချာစေပြီး (ကွန်တိန်နာ စတင်ရုံမျှမကဘဲ) အမှန်တကယ် အသင့်ဖြစ်မှုကို စောင့်ဆိုင်းစေသည်
- စိတ်ကြိုက် အပိုင်းများပါရှိသော **`networks`** သည် ဝန်ဆောင်မှုများကို သီးသန့်ခွဲထုတ်ပေးသည် — ဒေတာဘေ့စ်များကို backend ဂျွန်ယက်မှသာ ဝင်ရောက်နိုင်သင့်သည်
- **`volumes`** တွင် Named volumes သည် ဒေတာဘေ့စ် အချက်အလက်များကို ထိန်းသိမ်းပေးပြီး bind mounts (`./src:/app/src`) သည် development တွင် hot reload ကို ဖြစ်ပေါ်စေသည်
- **`restart: unless-stopped`** နှင့် အလှည့်ကျစနစ် (rotation) ပါရှိသော **`logging`** သည် production အတွက် အသင့်ဖြစ်နေသော ကွန်တိန်နာ စီမံခန့်ခွဲမှုကို ဖြစ်စေသည်

နောင်လာမည့် သင်ခန်းစာတွင်, သုံးထပ်ကွမ်း (three-tier) full-stack အပလီကေးရှင်း စတက်ခ်တစ်ခုကို အဆင့်ချင်း တည်ဆောက်သွားမည် ဖြစ်သည်။
